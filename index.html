async function register() { const nicknameInput = document.getElementById("registerNickname"); const emailInput = document.getElementById("registerEmail"); const passwordInput = document.getElementById("registerPassword"); const password2Input = document.getElementById("registerPassword2"); const message = document.getElementById("registerMessage"); const nickname = nicknameInput.value.trim(); const email = emailInput.value.trim(); const password = passwordInput.value; const password2 = password2Input.value; // 메시지 초기화 message.className = "message"; message.textContent = ""; // 닉네임 검사 if (!nickname) { message.className = "message error"; message.textContent = "닉네임을 입력해주세요."; nicknameInput.focus(); return; } if (nickname.length < 2) { message.className = "message error"; message.textContent = "닉네임은 2자 이상 입력해주세요."; nicknameInput.focus(); return; } if (nickname.length > 20) { message.className = "message error"; message.textContent = "닉네임은 20자 이하로 입력해주세요."; nicknameInput.focus(); return; } // 이메일 검사 if (!email) { message.className = "message error"; message.textContent = "이메일을 입력해주세요."; emailInput.focus(); return; } const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/; if (!emailPattern.test(email)) { message.className = "message error"; message.textContent = "올바른 이메일 주소를 입력해주세요."; emailInput.focus(); return; } // 비밀번호 검사 if (!password) { message.className = "message error"; message.textContent = "비밀번호를 입력해주세요."; passwordInput.focus(); return; } if (password.length < 6) { message.className = "message error"; message.textContent = "비밀번호는 6자 이상이어야 합니다."; passwordInput.focus(); return; } // 비밀번호 확인 if (!password2) { message.className = "message error"; message.textContent = "비밀번호 확인을 입력해주세요."; password2Input.focus(); return; } if (password !== password2) { message.className = "message error"; message.textContent = "비밀번호가 서로 다릅니다."; password2Input.focus(); return; } // 가입 처리 const { data, error } = await db.auth.signUp({ email: email, password: password, options: { data: { nickname: nickname } } }); // Supabase 오류 if (error) { message.className = "message error"; const errorText = error.message.toLowerCase(); if ( errorText.includes("already") || errorText.includes("registered") ) { message.textContent = "이미 가입된 이메일입니다."; } else if ( errorText.includes("password") ) { message.textContent = "비밀번호 조건을 확인해주세요."; } else if ( errorText.includes("email") ) { message.textContent = "이메일 주소를 확인해주세요."; } else { message.textContent = "회원가입에 실패했습니다. 입력 내용을 확인해주세요."; } // 중요: // 입력창은 그대로 유지한다. return; } if (!data || !data.user) { message.className = "message error"; message.textContent = "회원가입에 실패했습니다."; return; } // 프로필 생성 const { error: profileError } = await db .from("profiles") .insert({ id: data.user.id, nickname: nickname, badges: [], is_admin: false }); if (profileError) { message.className = "message error"; message.textContent = "계정은 만들어졌지만 프로필 생성에 실패했습니다. 관리자에게 문의해주세요."; return; } // 성공 message.className = "message success"; message.textContent = "회원가입이 완료되었습니다!"; setTimeout(() => { closeModals(); // 로그인 창으로 이동 openLogin(); // 이메일 자동 입력 document.getElementById( "loginEmail" ).value = email; }, 1000); }
